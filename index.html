CALCULATEUR

Moyenne

Calcule facilement ta moyenne scolaire.

function calculerMoyenne() { const devoir1 = parseFloat(document.getElementById("devoir1").value); const devoir2 = parseFloat(document.getElementById("devoir2").value); const composition = parseFloat(document.getElementById("composition").value); if ( isNaN(devoir1) || isNaN(devoir2) || isNaN(composition) ) { document.getElementById("resultatMoyenne").textContent = "Remplis toutes les notes."; return; } const moyenneDevoirs = (devoir1 + devoir2) / 2; const moyenneMatiere = (moyenneDevoirs + composition) / 2; document.getElementById("resultatMoyenne").textContent = "Ta moyenne : " + moyenneMatiere.toFixed(2) + " / 20"; }*

Révisions

Ajoute ce que tu dois réviser.

function ajouterRevision() { const matiere = document.getElementById("matiereRevision").value.trim(); const chapitre = document.getElementById("chapitreRevision").value.trim(); if (matiere === "" || chapitre === "") { alert("Remplis la matière et le chapitre."); return; } const revision = document.createElement("div"); revision.className = "task"; revision.innerHTML = ` ${matiere} — ${chapitre} `; document .getElementById("listeRevision") .appendChild(revision); document.getElementById("matiereRevision").value = ""; document.getElementById("chapitreRevision").value = ""; } function terminerRevision(checkbox) { const texte = checkbox.nextElementSibling; if (checkbox.checked) { texte.style.textDecoration = "line-through"; texte.style.opacity = "0.5"; } else { texte.style.textDecoration = "none"; texte.style.opacity = "1"; } } function supprimerRevision(bouton) { bouton.parentElement.remove(); }
